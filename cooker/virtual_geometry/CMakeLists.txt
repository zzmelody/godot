# This library is only used by Cooker and the value-only interface target.
# The product's editor/templates never link METIS.
set(VEYA_METIS_DIR "${CMAKE_CURRENT_LIST_DIR}/../../thirdparty/metis")
file(READ "${VEYA_METIS_DIR}/sources.json" VEYA_METIS_MANIFEST)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${VEYA_METIS_DIR}/sources.json")
set(VEYA_METIS_SOURCES)
foreach(component IN ITEMS metis gklib)
    string(JSON source_count LENGTH "${VEYA_METIS_MANIFEST}" "${component}_sources")
    math(EXPR source_last "${source_count} - 1")
    foreach(index RANGE 0 ${source_last})
        string(JSON source_name GET "${VEYA_METIS_MANIFEST}" "${component}_sources" ${index})
        if(component STREQUAL "metis")
            list(APPEND VEYA_METIS_SOURCES "${VEYA_METIS_DIR}/libmetis/${source_name}")
        else()
            list(APPEND VEYA_METIS_SOURCES "${VEYA_METIS_DIR}/gklib/${source_name}")
        endif()
    endforeach()
endforeach()
add_library(veya_metis STATIC ${VEYA_METIS_SOURCES})
target_include_directories(veya_metis SYSTEM PUBLIC "${VEYA_METIS_DIR}/include"
    PRIVATE "${VEYA_METIS_DIR}/libmetis" "${VEYA_METIS_DIR}/gklib")
target_compile_definitions(veya_metis PUBLIC IDXTYPEWIDTH=64
    PRIVATE USE_GKRAND USE_GKREGEX NDEBUG NDEBUG2)
if(MSVC)
    target_compile_definitions(veya_metis PRIVATE WIN32 _CRT_SECURE_NO_WARNINGS "__thread=__declspec(thread)")
    target_compile_options(veya_metis PRIVATE /w)
else()
    target_compile_options(veya_metis PRIVATE -w -fno-strict-aliasing)
    target_link_libraries(veya_metis PUBLIC m)
endif()
find_package(Threads REQUIRED)
add_library(veya_geometry_partition STATIC graph_partitioner.cpp graph_locality.cpp triangle_topology.cpp leaf_clusters.cpp attribute_quadric.cpp attribute_policy.cpp mesh_simplifier.cpp cluster_dag.cpp cluster_stripifier.cpp cluster_encoding.cpp ../../core/io/virtual_geometry/cluster_codec.cpp ../../core/io/virtual_geometry/topology_codec.cpp ../../core/io/virtual_geometry/attribute_codec.cpp ../../core/io/virtual_geometry/page_codec.cpp)
target_include_directories(veya_geometry_partition PUBLIC "${CMAKE_CURRENT_LIST_DIR}/../..")
target_sources(veya_geometry_partition PRIVATE page_builder.cpp metadata_bounds.cpp hierarchy_builder.cpp model_builder.cpp ../../core/io/virtual_geometry/hierarchy.cpp ../../core/io/virtual_geometry/model_codec.cpp ../../core/io/virtual_geometry/runtime_fixups.cpp ../../core/io/virtual_geometry_page_directory.cpp)
target_link_libraries(veya_geometry_partition PRIVATE veya_metis Threads::Threads)
